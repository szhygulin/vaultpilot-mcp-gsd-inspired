2. **`src/signing/morpho-shares-math.ts` (NEW).** Pure-bigint module. Per research § Topic 4. Mirror `src/signing/aave-health.ts` shape (constants + functions + pure-bigint discipline; no I/O, no side effects).
   ```typescript
   /**
    * SharesMathLib — pure-bigint shares↔assets conversion mirroring
    * morpho-blue's SharesMathLib.sol (research § Topic 4 + § Don't Hand-Roll).
    *
    * VIRTUAL_SHARES + VIRTUAL_ASSETS are anti-inflation guards baked into the
    * protocol's accounting — drift here = misalignment with on-chain reality.
    * Constants pinned at write time; regression-tested.
    *
    * Used by:
    *   - src/chains/morpho-blue.ts::computeExpectedSupplyAssets/borrowAssets
    *   - Plan 29-03 prepare_morpho_repay (if needed for display-side debt rendering)
    */
   export const VIRTUAL_SHARES = 1_000_000n;  // 1e6 — research § Topic 4
   export const VIRTUAL_ASSETS = 1n;          // research § Topic 4

   export function toAssetsDown(shares: bigint, totalAssets: bigint, totalShares: bigint): bigint {
     return (shares * (totalAssets + VIRTUAL_ASSETS)) / (totalShares + VIRTUAL_SHARES);
   }

   export function toAssetsUp(shares: bigint, totalAssets: bigint, totalShares: bigint): bigint {
     return (
       (shares * (totalAssets + VIRTUAL_ASSETS) + totalShares + VIRTUAL_SHARES - 1n) /
       (totalShares + VIRTUAL_SHARES)
     );
   }

   /** Inverse — assets → shares; rounded down (conservative; matches SharesMathLib.toSharesDown). */
   export function toSharesDown(assets: bigint, totalAssets: bigint, totalShares: bigint): bigint {
     return (assets * (totalShares + VIRTUAL_SHARES)) / (totalAssets + VIRTUAL_ASSETS);
   }

   export function toSharesUp(assets: bigint, totalAssets: bigint, totalShares: bigint): bigint {
     return (
       (assets * (totalShares + VIRTUAL_SHARES) + totalAssets + VIRTUAL_ASSETS - 1n) /
       (totalAssets + VIRTUAL_ASSETS)
     );
   }
   ```
   The inverse direction (`toSharesDown` / `toSharesUp`) is shipped for completeness even though Plan 29-02 only uses `toAssetsDown` / `toAssetsUp`; Plan 29-03 may use the inverse for prepare-time encoding if a share-based path emerges (currently only repay-max is share-based, using shares read directly from position()).

   ```typescript
     const [totalSupplyAssets, totalSupplyShares, totalBorrowAssets, totalBorrowShares, lastUpdate, fee] =
       await client.readContract({
         address: morpho,
         abi: MORPHO_BLUE_ABI,
         functionName: "market",
         args: [marketId],
       });
     return { totalSupplyAssets, totalSupplyShares, totalBorrowAssets, totalBorrowShares, lastUpdate, fee };
   }

   /** Intent-vs-reality gate input — Plan 29-03 consumes. */

     const [loanToken, collateralToken, oracle, irm, lltv] = await client.readContract({
       address: morpho,
       abi: MORPHO_BLUE_ABI,
       functionName: "idToMarketParams",
       args: [marketId],
     });
     return { loanToken, collateralToken, oracle, irm, lltv };
   }

   /** Event-log scan for market discovery — research § Pattern 3. */
   async function scanTouchedMarkets(client: PublicClient, wallet: Address): Promise<Set<Hex>> {
     const morpho = getMorphoBlueAddress(1)!;
     const currentBlock = await client.getBlockNumber();
     const LOOKBACK = 1_000_000n;  // 1M blocks ≈ 140 days; mirror Phase 8 DF-2 default
     const fromBlock = currentBlock > LOOKBACK ? currentBlock - LOOKBACK : 0n;
     // Chunked getLogs (100k-block chunks per Phase 8 pattern) — omitted for brevity here;
     // implementation iterates fromBlock → currentBlock in 100k-block chunks per event type.
     const [supplyLogs, borrowLogs, collateralLogs] = await Promise.all([
       client.getLogs({
         address: morpho,
         event: MORPHO_BLUE_ABI[/* Supply event index */],
         args: { onBehalf: getAddress(wallet) },
         fromBlock,
       }),
       client.getLogs({
         address: morpho,
         event: MORPHO_BLUE_ABI[/* Borrow event index */],
         args: { onBehalf: getAddress(wallet) },
         fromBlock,
       }),
       client.getLogs({
         address: morpho,
         event: MORPHO_BLUE_ABI[/* SupplyCollateral event index */],
         args: { onBehalf: getAddress(wallet) },
         fromBlock,
       }),
     ]);
     // Set of unique marketIds — research § Pattern 3 final deduplication
     return new Set([
       ...supplyLogs.map(l => l.args.id as Hex),
       ...borrowLogs.map(l => l.args.id as Hex),
       ...collateralLogs.map(l => l.args.id as Hex),
     ]);
   }

   /** Off-chain accrual simulation — supply side (rounded down). */
   function computeExpectedSupplyAssets(
     pos: { supplyShares: bigint },
     mkt: { totalSupplyAssets: bigint; totalSupplyShares: bigint }
   ): bigint {
     return toAssetsDown(pos.supplyShares, mkt.totalSupplyAssets, mkt.totalSupplyShares);
   }

   /** Off-chain accrual simulation — borrow side (rounded UP — conservative for debt display). */
   function computeExpectedBorrowAssets(
     pos: { borrowShares: bigint },
     mkt: { totalBorrowAssets: bigint; totalBorrowShares: bigint }
   ): bigint {
     return toAssetsUp(pos.borrowShares, mkt.totalBorrowAssets, mkt.totalBorrowShares);
   }

   export { readPosition, readMarket, readMarketParams, scanTouchedMarkets, computeExpectedSupplyAssets, computeExpectedBorrowAssets };

   /** ESM spy indirection — CLAUDE.md Conventions; Plan 29-03 consumes via `vi.spyOn(_morphoChains, "<fn>")`. */

     readPosition,
     readMarket,
     readMarketParams,
     scanTouchedMarkets,
     computeExpectedSupplyAssets,
     computeExpectedBorrowAssets,
   };
   ```
