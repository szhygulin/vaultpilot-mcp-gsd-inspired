## `src/cli/setup-mcp-clients.ts` shape (LOCKED — RESEARCH § Topic 6 lines 477-531 verbatim; SHARED with Plan 10-02)

```typescript
// src/cli/setup-mcp-clients.ts (NEW Phase 10 Plan 10-03 — SHARED with Plan 10-02 install.sh consumer)
//
// Per-platform MCP client detection + JSON merge. Plan 10-03 OWNS this file;
// Plan 10-02 install.sh imports detectMcpClients + registerWithJsonConfig
// via the binary path (vaultpilot-mcp setup --register-clients-only).

import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { homedir, platform } from "node:os";
import { execSync } from "node:child_process";

export interface DetectedClient {
  name: "claude-code" | "claude-desktop" | "cursor";
  configPath: string | null;  // null for claude-code (CLI-managed via `claude mcp add`)
  detected: boolean;
}

function commandExists(cmd: string): boolean {
  try { execSync(`command -v ${cmd}`, { stdio: "ignore" }); return true; }
  catch { return false; }
}

export function detectMcpClients(): DetectedClient[] {
  const home = homedir();
  const plat = platform();  // "darwin" | "linux" | "win32"
  const claudeDesktopPath =
    plat === "darwin" ? join(home, "Library", "Application Support", "Claude", "claude_desktop_config.json") :
    plat === "win32"  ? join(process.env.APPDATA ?? "", "Claude", "claude_desktop_config.json") :
                        join(home, ".config", "Claude", "claude_desktop_config.json");
  const cursorPath = join(home, ".cursor", "mcp.json");
  return [
    { name: "claude-code",    configPath: null,                 detected: commandExists("claude") },
    { name: "claude-desktop", configPath: claudeDesktopPath,    detected: existsSync(claudeDesktopPath) },
    { name: "cursor",         configPath: cursorPath,           detected: existsSync(cursorPath) },
  ];
}

export async function registerWithJsonConfig(
  configPath: string,
  binaryPath: string,
  wrapForWindows: boolean,
): Promise<void> {
  let existing: { mcpServers?: Record<string, unknown> } = {};
  if (existsSync(configPath)) {
    const raw = readFileSync(configPath, "utf8");
    try { existing = JSON.parse(raw); }
    catch { throw new Error(`Refused — ${configPath} is malformed JSON. Fix the file first, then re-run \`vaultpilot-mcp setup\`.`); }
  }
  const serverEntry = wrapForWindows
    ? { command: "cmd", args: ["/c", binaryPath] }
    : { command: binaryPath };
  const merged = {
    ...existing,
    mcpServers: { ...(existing.mcpServers ?? {}), "vaultpilot-mcp": serverEntry },
  };
  writeFileSync(configPath, JSON.stringify(merged, null, 2) + "\n");
}

export function registerWithClaudeCode(binaryPath: string): void {
  try { execSync(`claude mcp remove vaultpilot-mcp`, { stdio: "ignore" }); } catch { /* best-effort */ }
  execSync(`claude mcp add vaultpilot-mcp -- ${binaryPath}`, { stdio: "inherit" });
}
```

export async function runSetup(args: string[]): Promise<number> {
  const nonInteractive = args.includes("--non-interactive");
  const dryRun = args.includes("--dry-run");
  const jsonMode = args.includes("--json");
