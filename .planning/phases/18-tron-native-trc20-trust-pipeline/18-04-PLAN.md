      ```typescript
      // Phase 18 — Plan 18-04 — TRON branch (additive; lives OUTSIDE the FROZEN
      // three-gate region above). Dispatcher in the main handler reads
      // `record.tx.txType` and routes here for `"tron"` handles. The EVM + Solana
      // branches above stay byte-frozen.
      //
      // TRON branch layers (per CONTEXT D-03 + D-11):
      //   Layer 0.5 — canonical-dispatch-tron (TRC-20 only; native skips)
      //   Layer 0.7 — TRC-20 mandatory `triggerconstantcontract` refusal
      //              native: emitNoSimulationAvailable() advisory (NOT refusal)
      //   Layer 1   — handle lookup (done at dispatcher; passed in as `record`)
      //   Layer 3   — fingerprint drift caught at `send_transaction.ts` PAYLOAD_FINGERPRINT_DRIFT gate
      async function previewSendTron(
        record: HandleRecord & { tx: PreparedTxTron },
        args: PreviewSendArgs,
        requestContext: RequestContext,
      ): Promise<ToolResponse> {
        // Layer 0.5 — canonical-dispatch-tron (TRC-20 only)
        if (record.tx.kind === "trc20") {
          const result = _canonicalDispatchTron.checkTronDispatchTarget([
            record.tx.contractAddress!,
          ]);
          if (result.kind === "refused") {
            return errEnvelope("DISPATCH_TARGET_REFUSED", {
              message: `TRC-20 contract ${record.tx.contractAddress} is not in the canonical-dispatch-tron allowlist (4-entry stablecoin set per Phase 18 / TRON-PREP-02). Use one of: ${result.allowlist.join(", ")}. Non-allowlist tokens land at Phase 19+.`,
              cause: { offenders: result.offenders, allowlist: result.allowlist },
            });
          }
        }
        // Native TRX skips Layer 0.5 per CONTEXT D-11a.

        // Layer 0.7 — TRC-20 mandatory simulation gate; native advisory
        let simulationResult: TronSimulationResult;
        if (record.tx.kind === "trc20") {
          const tronWeb = _tronRegistry.getTronWeb();
          // Reconstruct simulation inputs from instructionSummary + raw_data:
          //   contractAddress = record.tx.contractAddress (already validated at Layer 0.5)
          //   functionSelector = "transfer(address,uint256)" (single TRC-20 function in Phase 18)
          //   parameters = [{ type: "address", value: instructionSummary[0].to }, { type: "uint256", value: instructionSummary[0].amount.toString() }]
          //   ownerAddress = instructionSummary[0].from (the prepare-time sender)
          const summary = record.tx.instructionSummary![0];
          if (summary.kind !== "trc20-transfer") {
            return errEnvelope("INTERNAL_ERROR", {
              message: "Handle's instructionSummary kind mismatch — expected 'trc20-transfer'",
            });
          }
          simulationResult = await _simulationTron.runTronPreviewSimulation({
            tronWeb,
            contractAddress: record.tx.contractAddress!,
            functionSelector: "transfer(address,uint256)",
            parameters: [
              { type: "address", value: summary.to },
              { type: "uint256", value: summary.amount.toString() },
            ],
            ownerAddress: summary.from,
          });
          if (simulationResult.status !== "ok") {
            return errEnvelope("SIMULATION_REFUSED", {
              message: `TRC-20 preview simulation refused: ${simulationResult.status}${simulationResult.revertReason ? ` (reason: ${simulationResult.revertReason})` : ""}`,
              cause: simulationResult,
            });
          }
        } else {
          simulationResult = _simulationTron.emitNoSimulationAvailable();
          // status: "not-applicable" — NOT a refusal. Per CONTEXT D-03b.
        }

        // Recompute presign hash for LEDGER BLIND-SIGN HASH block
        const rawDataBytes = Buffer.from(record.tx.rawDataHex, "hex");
        const { presignHash } = _tronPresign.computeTronPresignHash({
          rawDataBytes: new Uint8Array(rawDataBytes),
        });

        // Pin PreviewPinned (mirrors Solana sentinel-zeros for EVM-shape fields)
        const previewToken = crypto.randomUUID();
        const pinned: PreviewPinned = {
          nonce: 0,                  // sentinel
          gas: 0n,                   // sentinel
          maxFeePerGas: 0n,          // sentinel
          maxPriorityFeePerGas: 0n,  // sentinel
          previewToken,
          presignHash,
          selector: record.tx.kind === "trc20" ? "0xa9059cbb" : null,
        };
        transitionToPreviewed(record.handle, pinned);

        // Build text response blocks
        const prepareReceipt = record.tx.kind === "native"
          ? PREPARE_RECEIPT_TRON_NATIVE_TEMPLATE
              .replace("{TO}", record.args.to)
              .replace("{SUN}", record.args.sun!)
              .replace("{REF_BLOCK_BYTES}", record.tx.refBlockBytes)
              .replace("{REF_BLOCK_HASH}", record.tx.refBlockHash)
              .replace("{EXPIRATION}", String(record.tx.expiration))
          : PREPARE_RECEIPT_TRON_TRC20_TEMPLATE
              .replace("{TO}", record.args.to)
              .replace("{TOKEN_ADDRESS}", record.args.tokenAddress!)
              .replace("{AMOUNT}", record.args.amount!)
              .replace("{REF_BLOCK_BYTES}", record.tx.refBlockBytes)
              .replace("{REF_BLOCK_HASH}", record.tx.refBlockHash)
              .replace("{EXPIRATION}", String(record.tx.expiration));

        const blindSignHashBlock = LEDGER_BLIND_SIGN_HASH_TRON_TEMPLATE
          .replace("{HASH_FULL_64HEX}", presignHash)
          .replace("{HASH_CHUNKED_4_CHAR_GROUPS}", chunkHex(presignHash));

        const simulationBlock = record.tx.kind === "trc20"
          ? SIMULATION_BLOCK_TRON_TEMPLATE
              .replace("{STATUS}", simulationResult.status)
              .replace("{REVERT_REASON}", simulationResult.revertReason ?? "")
              .replace("{ENERGY_USED}", simulationResult.energyUsed?.toString() ?? "n/a")
              .replace("{CONSTANT_RESULT_PREVIEW}", simulationResult.constantResult.slice(0, 1).join(""))
          : NO_SIMULATION_AVAILABLE_TRON_TEMPLATE;

        // LEDGER_NOTICE_TRON_TEMPLATE — emit conditionally via predicate function.
        // Phase 18 stablecoin set is in TRX app v0.5+ bundled registry per
        // CONTEXT D-04b — predicate returns `{ emit: false }`. Phase 19+ widens
        // the predicate when TRC-20 approve / Stake 2.0 / SunSwap router calls
        // land (those are NOT in the bundled clear-sign registry).
        const noticeDecision = shouldEmitTronLedgerNotice(record.tx);
        const ledgerNoticeBlock = noticeDecision.emit
          ? LEDGER_NOTICE_TRON_TEMPLATE
              .replace("{INSTRUCTION_NAME}", noticeDecision.instructionName)
              .replace("{REGISTRY_STATUS}", noticeDecision.reason)
          : "";

        const verifyBeforeSigning = VERIFY_BEFORE_SIGNING_TRON_TEMPLATE;

        const responseText = [
          prepareReceipt,
          ledgerNoticeBlock,
          blindSignHashBlock,
          simulationBlock,
          verifyBeforeSigning,
          `\nPreview token: ${previewToken}`,
          `\nNext step: send_transaction({ handle: "${record.handle}", previewToken: "${previewToken}", userDecision: "send" })`,
        ].filter(Boolean).join("\n\n");

        return {
          content: [{ type: "text", text: responseText }],
          structuredContent: {
            handle: record.handle,
            chain: "tron",
            kind: record.tx.kind,
            previewToken,
            presignHash,
            simulation: simulationResult,
            payloadFingerprint: record.payloadFingerprint,
            decodedArgs: record.tx.instructionSummary,
            blockHeader: {
              refBlockBytes: record.tx.refBlockBytes,
              refBlockHash: record.tx.refBlockHash,
              expiration: record.tx.expiration,
            },
            rawDataHex: record.tx.rawDataHex,
          },
        };
      }
      ```

   c. **New sibling function `sendTransactionTron(record, args, requestContext)`** below the Solana branch (after `// Phase 12 — Plan 12-05 — Solana branch` block; approximately line ~1010+):

      ```typescript
      // Phase 18 — Plan 18-04 — TRON branch (additive; lives OUTSIDE the FROZEN
      // three-gate region above). The three FROZEN gates (PREVIEW_REQUIRED +
      // PREVIEW_TOKEN_MISMATCH + PAYLOAD_FINGERPRINT_DRIFT) fired identically
      // before this branch was entered. Demo-mode branch (DEMO-05) is
      // BYTE-FROZEN with the EVM + Solana arms above.
      //
      // TRON broadcast path:
      //   1. Open USB-HID transport via _tronLedgerTransport.signTransaction
      //      (Phase 17 wired — per-call open + try/finally close)
      //   2. Wrap signature into outer Transaction envelope (signature[] field)
      //   3. Broadcast via tronweb.trx.sendRawTransaction(signedTransaction)
      //   4. Stamp handle via transitionToSent
      //
      // Error envelope mapping (RESEARCH §Topic 7):
      //   - Open failure → LEDGER_NOT_CONNECTED (errorCode 17 — reuse from Phase 12)
      //   - User rejects on-device → LEDGER_REJECTED (errorCode 8 — original v1.x)
      //   - Broadcast result.code === "TRANSACTION_EXPIRATION_ERROR" → BROADCAST_FAILED
      //   - Broadcast result.code === "SIGERROR" → LEDGER_REJECTED (signature didn't verify)
      //   - Broadcast result.code === "BANDWITH_ERROR" / "CONTRACT_VALIDATE_ERROR" → BROADCAST_FAILED

      async function sendTransactionTron(
        record: HandleRecord & { tx: PreparedTxTron },
        args: SendTransactionArgs,
        requestContext: RequestContext,
      ): Promise<ToolResponse> {
        // Demo mode (DEMO-05) — simulate via triggerconstantcontract for TRC-20;
        // native skips with `[DEMO MODE]` advisory + no broadcast.
        if (getDemoState().mode === "demo") {
          // Return simulation envelope per Phase 5 DEMO-05 pattern
          return buildTronDemoSimulationResponse(record);
        }

        // Real mode — sign via Ledger TRX app
        let signature: string;
        try {
          const account = await getActiveNonEvmAccount("tron");
          if (!account) {
            return errEnvelope("WALLET_NOT_PAIRED", {
              message: "No paired TRON Ledger account; call pair_tron_ledger first.",
            });
          }
          signature = await _tronLedgerTransport.signTransaction({
            path: account.derivationPath,
            rawTxHex: record.tx.rawDataHex,
            tokenSignatures: [],  // Phase 18 — bundled token registry covers Phase 18 set
          });
        } catch (err) {
          // Map transport errors to structured envelope (Phase 17 patterns)
          if (err instanceof LedgerDeviceNotConnectedError) {
            return errEnvelope("LEDGER_NOT_CONNECTED", { message: err.message });
          }
          if (err instanceof LedgerTronAppNotOpenError) {
            // Reuse Phase 12's SOLANA_APP_NOT_OPEN code? NO — per error-codes BYTE-FROZEN.
            // Re-purpose existing code: LEDGER_REJECTED with cause naming the wrong-app state.
            return errEnvelope("LEDGER_REJECTED", {
              message: "Ledger TRX app not open. Open the TRX app on the device and retry.",
              cause: err.message,
            });
          }
          // User-rejected-on-device → Ledger returns a specific error
          if (err instanceof Error && /reject/i.test(err.message)) {
            return errEnvelope("LEDGER_REJECTED", { message: err.message });
          }
          return errEnvelope("INTERNAL_ERROR", {
            message: "TRON Ledger signing failed",
            cause: err instanceof Error ? err.message : String(err),
          });
        }

        // Wrap signature into outer envelope per RESEARCH §Topic 7.
        // CRITICAL: `txID` field is TRON's SHA-256(raw_data) — that is
        // `record.pinned.presignHash` (computed by previewSendTron above), NOT
        // `record.payloadFingerprint` (which is keccak256(domain-tag ‖ raw_data) —
        // the VaultPilot binding hash, distinct from TRON consensus tx-id).
        // See RESEARCH §Topic 4 for the hash-function-per-layer rationale.
        const signedTransaction = {
          visible: true,
          txID: record.pinned!.presignHash.slice(2),  // strip 0x prefix; TRON tx-id == SHA-256(raw_data) hex
          raw_data: record.tx.rawDataObject,
          raw_data_hex: record.tx.rawDataHex,
          signature: [signature],
        };

        // Broadcast via tronweb
        const tronWeb = _tronRegistry.getTronWeb();
        let broadcastResult;
        try {
          broadcastResult = await tronWeb.trx.sendRawTransaction(signedTransaction);
        } catch (err) {
          return errEnvelope("BROADCAST_FAILED", {
            message: "TronGrid broadcast call failed",
            cause: err instanceof Error ? err.message : String(err),
          });
        }

        // Map broadcast envelope to structured response
        if (broadcastResult.result !== true) {
          const code = broadcastResult.code ?? "UNKNOWN_CODE";
          if (code === "SIGERROR") {
            return errEnvelope("LEDGER_REJECTED", {
              message: "Signature did not verify against the public key — transport corruption suspected",
              cause: broadcastResult.message,
            });
          }
          return errEnvelope("BROADCAST_FAILED", {
            message: `TronGrid refused broadcast: ${code}`,
            cause: broadcastResult.message,
          });
        }

        // Success — stamp handle, return tx hash
        const txHash = broadcastResult.txid;  // hex string, no 0x prefix per TRON convention
        transitionToSent(record.handle, txHash);

        return {
          content: [{
      ```

   ```markdown
   ## TRON (v2.1 — Phase 18)

   The v2.1 TRON trust pipeline mirrors v2.0's Solana shape with TRON-specific cryptographic-binding primitives. The same Layer 1 / Layer 2 / Layer 3 defenses apply identically; the per-chain primitives differ at the binding layer.

   ### (1) USB-HID transport trust shape

   TRON signing uses `@ledgerhq/hw-app-trx` over `@ledgerhq/hw-transport-node-hid` — same USB-HID transport as Phase 12 Solana, no WalletConnect bridge. Per-call transport open + `try/finally` close (mirrors Phase 12 + Phase 17 patterns). The agent → MCP path runs over stdio; the MCP → Ledger path runs over USB-HID. The Ledger device's physical screen is the trust anchor.

   ### (2) Protobuf raw_data preimage vs EVM RLP

   TRON's transaction binding hash runs over Protobuf-serialized `raw_data` bytes (`tronweb.transaction.raw_data_hex` → `Buffer.from(hex, "hex")`). This is structurally distinct from EVM's RLP-serialized EIP-1559 envelope: TRON's bytes describe an array of contract-instructions with explicit type discriminators (`TransferContract`, `TriggerSmartContract`, etc.); EVM's bytes are a single-tuple `[chainId, nonce, maxPriorityFee, maxFee, gas, to, value, data, accessList]`. Both hash via keccak256 at the VaultPilot binding layer; the domain tag (`"VaultPilot-trontx-v1:"` — 21 UTF-8 bytes, distinct from EVM 23 + Solana 20) makes cross-chain reuse impossible by construction.

   ### (3) SHA-256 tx-id (TRON consensus) vs EVM keccak256

   TRON consensus computes `tx_id = SHA-256(raw_data)` (NOT keccak256 — TRON predates Ethereum's RLP-keccak combo and uses a distinct hash). The Ledger TRX app displays this exact 32-byte SHA-256 hash on blind-sign mode, labeled `"Transaction ID"`. The `LEDGER BLIND-SIGN HASH (TRON)` block in `preview_send` shows the same hash for character-for-character on-device comparison. Within the cryptographic-binding chain: `payloadFingerprint = keccak256(domain-tag ‖ raw_data)` is the agent↔MCP binding; `ledgerBlindSignHash = SHA-256(raw_data)` is the MCP↔device binding. Same input bytes; different hash functions for layer-appropriate uniqueness.

   ### (4) TRX app clear-sign coverage + bundled token registry

   The Ledger TRX app v0.5+ ships a bundled token registry that clear-signs `transfer(address,uint256)` calldata for the bundled tokens: USDT-TRC20, USDC-TRC20, USDD, TUSD (the Phase 18 allowlist). For these tokens, the device displays `To`, `Token`, `Amount` decoded — the user sees the same data the PREPARE RECEIPT block surfaced server-side. For non-bundled tokens (Phase 19+), the app falls back to blind-sign mode (displays only the SHA-256 tx-id); Phase 18 + future phases emit `LEDGER NOTICE (TRON)` block above the BLIND-SIGN HASH block to warn the user.

   `prepare_tron_token_approve` (Phase 19 — TRC-20 `approve(spender, amount)`) is a distinct ABI from `transfer` and is NOT in the bundled registry; the conditional NOTICE fires for every approve. `prepare_tron_stake_*` (Phase 19 — Stake 2.0 FreezeBalanceV2Contract) is a different Protobuf shape entirely from TriggerSmartContract; the app blind-signs the consensus tx-id and the NOTICE fires. `prepare_sunswap_*` (Phase 20 — SunSwap router calls) are TriggerSmartContract calls to NON-allowlist contracts; the NOTICE fires.

   ### (5) Layer 0.7 asymmetry — TRC-20 mandatory refusal, native TRX advisory only (accepted residual)

   The Layer 0.7 simulation gate behaves asymmetrically between TRC-20 and native TRX. **TRC-20**: `preview_send` calls `triggerconstantcontract` against TronGrid and refuses with `SIMULATION_REFUSED` if the simulated execution reverts. **Native TRX (`TransferContract`)**: TronGrid has no simulation API for non-contract calls — the `triggerconstantcontract` endpoint is for smart-contract `view`/`pure` calls only. `preview_send` emits an explicit `CHECKS PERFORMED (TRON — no simulation available)` block instead of refusing.

   **Accepted residual risk.** A native TRX transfer's defense at preview time relies entirely on (1) PREPARE RECEIPT byte-binding the agent's args verbatim + (2) LEDGER BLIND-SIGN HASH on-device match. A compromised agent that swaps the `to` address between the user's natural-language ask and the prepare call would be caught by PREPARE RECEIPT (the user reads the verbatim args in the response); a compromised MCP that swaps bytes between prepare and send would be caught by `payloadFingerprint` drift detection (PREP-08, Layer 3). The on-device tx-id is the final anchor. The asymmetry is surfaced visibly to the user via the `NO_SIMULATION_AVAILABLE_TRON` block — accepted residual; documented residual.

   ### (6) Ref-block + expiration window — TAPOS replay protection + extended expiration

   TRON transactions carry two time-bounded fields:
   - `ref_block_bytes` + `ref_block_hash` — TAPOS (Transactions-as-Proof-of-Stake) replay protection. The transaction references the latest solidified block at build time; validators reject the transaction if the referenced block falls outside their recent-solidified-chain cache (~15-30 minutes empirically).
   - `expiration` — absolute millisecond timestamp at which the broadcast endpoint refuses the transaction. tronweb defaults to `block_timestamp + 60_000ms` (60 seconds).

   Phase 18 prepare tools extend the expiration to **900 seconds (15 minutes)** via `tronweb.transactionBuilder.extendExpiration(tx, 900)` to match the 15-minute handle TTL (`HANDLE_TTL_MS`). Without the extension, a user who pauses ~5 minutes between prepare and send would broadcast a stale transaction and receive `BROADCAST_FAILED: TRANSACTION_EXPIRATION_ERROR`.

   The 15-minute window sits inside the TAPOS replay window — both are valid for the same duration window. Past 15 minutes, the handle expires (`HANDLE_EXPIRED` envelope on `preview_send` / `send_transaction`), the user re-runs `prepare_tron_*` for a fresh handle with fresh ref-block + expiration fields. Defense-in-depth: the Layer 3 fingerprint-drift gate also catches any post-prepare in-process state corruption.
   ```
