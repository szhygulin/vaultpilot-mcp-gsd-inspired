// `CANONICAL_DISPATCH_TARGETS` table — NOT a widening of

// (DECODED ARGS approve template); dispatch-allowlist is a SECURITY concern

// Per-chain entries (4 per chain × 5 chains = 20 membership assertions; 12 unique

//   - Aave V3 Pool (per-chain via getAaveV3PoolAddress getter — Phase 7 SOT)

import { getAddress, type Address } from "viem";

import {
  getAaveV3PoolAddress,
  getWethAddress,
  type ChainId,
} from "../config/contracts.js";

const ONEINCH_V6_ROUTER_ALL_CHAINS: Address = getAddress(
  "0x111111125421cA6dc452d289314280a0F8842A65",
  // Verified at portal.1inch.dev for Ethereum/Arbitrum/Polygon/Base/Optimism per RESEARCH § A3 + KNOWN_SPENDERS_ETHEREUM row 3
);

const LIFI_DIAMOND_ALL_CHAINS: Address = getAddress(
  "0x1231DEB6f5749EF6cE6943a275A1D3E7486F4EaE",
  // Verified at LiFi docs for Ethereum/Arbitrum/Polygon/Base/Optimism per RESEARCH § A3 + KNOWN_SPENDERS_ETHEREUM row 2
);

function buildPerChainAllowlist(chainId: ChainId): ReadonlySet<Address> {
  return new Set<Address>([
    getAaveV3PoolAddress(chainId),
    getWethAddress(chainId),
    ONEINCH_V6_ROUTER_ALL_CHAINS,
    LIFI_DIAMOND_ALL_CHAINS,
  ]);
}

export const CANONICAL_DISPATCH_TARGETS: Readonly<Record<ChainId, ReadonlySet<Address>>> = {
  1:     buildPerChainAllowlist(1),
  42161: buildPerChainAllowlist(42161),
  137:   buildPerChainAllowlist(137),
  8453:  buildPerChainAllowlist(8453),
  10:    buildPerChainAllowlist(10),
};

export type DispatchCheckResult =
  | { kind: "ok" }
  | { kind: "refused"; chain: ChainId; to: Address; allowlist: Address[] };

export function checkDispatchTarget(chainId: ChainId, to: Address): DispatchCheckResult {
  const checksummed = getAddress(to);  // EIP-55 normalization — case-insensitive match
  const allowlist = CANONICAL_DISPATCH_TARGETS[chainId];
  if (allowlist.has(checksummed)) return { kind: "ok" };
  return { kind: "refused", chain: chainId, to: checksummed, allowlist: [...allowlist] };
}

export const _canonicalDispatch = { checkDispatchTarget };  // ESM spy-affordance per CLAUDE.md Convention
