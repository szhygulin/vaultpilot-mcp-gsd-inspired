      }
    }

        divergences.push({ field: "amount", agentSaid: claimedDecode.args.amount, serverSays: wethDecoded.amountWei.toString() });
      }
    }
  } else if (action === "aave-supply" && aaveDecoded.kind === "aave-supply") {
    const aaveExpected = getAaveV3PoolAddress(record.tx.chainId as ChainId);
    if (recordTo !== aaveExpected) divergences.push({ field: "to", agentSaid: recordTo, serverSays: `${aaveExpected} (canonical Aave V3 Pool for chain ${record.tx.chainId})` });
    if (checksummedClaimedTo !== aaveExpected) divergences.push({ field: "to (claimedDecode)", agentSaid: checksummedClaimedTo, serverSays: aaveExpected });
    if (claimedDecode.args.asset && getAddress(claimedDecode.args.asset) !== getAddress(aaveDecoded.asset)) {
      divergences.push({ field: "asset", agentSaid: getAddress(claimedDecode.args.asset), serverSays: getAddress(aaveDecoded.asset) });
    }
    if (claimedDecode.args.onBehalfOf && getAddress(claimedDecode.args.onBehalfOf) !== getAddress(aaveDecoded.onBehalfOf)) {
      divergences.push({ field: "onBehalfOf", agentSaid: getAddress(claimedDecode.args.onBehalfOf), serverSays: getAddress(aaveDecoded.onBehalfOf) });
    }
    if (claimedDecode.args.amount) {
      const claimedWei = parseAmountStrict(claimedDecode.args.amount, 18);
      if (claimedWei !== aaveDecoded.amountWei) {
        divergences.push({ field: "amount", agentSaid: claimedDecode.args.amount, serverSays: aaveDecoded.amountWei.toString() });
      }
    }
  } else if (action === "aave-withdraw" && aaveDecoded.kind === "aave-withdraw") {
    const aaveExpected = getAaveV3PoolAddress(record.tx.chainId as ChainId);
    if (recordTo !== aaveExpected) divergences.push({ field: "to", agentSaid: recordTo, serverSays: aaveExpected });
    if (checksummedClaimedTo !== aaveExpected) divergences.push({ field: "to (claimedDecode)", agentSaid: checksummedClaimedTo, serverSays: aaveExpected });
    if (claimedDecode.args.asset && getAddress(claimedDecode.args.asset) !== getAddress(aaveDecoded.asset)) {
      divergences.push({ field: "asset", agentSaid: getAddress(claimedDecode.args.asset), serverSays: getAddress(aaveDecoded.asset) });
    }
    if (claimedDecode.args.to && getAddress(claimedDecode.args.to) !== getAddress(aaveDecoded.to)) {
      divergences.push({ field: "to (recipient)", agentSaid: getAddress(claimedDecode.args.to), serverSays: getAddress(aaveDecoded.to) });
    }
    if (claimedDecode.args.amount) {
      const claimedWei = parseAmountStrict(claimedDecode.args.amount, 18);
      if (claimedWei !== aaveDecoded.amountWei) {
        divergences.push({ field: "amount", agentSaid: claimedDecode.args.amount, serverSays: aaveDecoded.amountWei.toString() });
      }
    }
  } else {
    // Action mismatch — agent claims an action the server's decoder doesn't recognize for this calldata

  }

  if (divergences.length === 0) {
    return {
      content: [{ type: "text" as const, text: "ok: agent's claimed decode matches server's independent decode" }],
      structuredContent: { kind: "ok" as const },
    };
  }
  return {
    content: [{ type: "text" as const, text: `divergence: ${divergences.length} field(s) disagree between agent claim and server decode` }],
    structuredContent: { kind: "divergence" as const, divergences },
  };
});

## Per-token decimals for parseAmountStrict TODO

The `parseAmountStrict(amount, decimals)` call in `verify_tx_decode` uses a HARDCODED `18` for decimals in the sketch above — this is a known LIMITATION because the actual decimals for an ERC-20 token vary (USDC = 6, DAI = 18, WBTC = 8, etc.). For a CORRECT comparison, the server would need to resolve decimals via `get_token_metadata` on the token contract. **Three options**:

- **(a) Always use 18 — accept that the comparison is approximate** for non-18-decimal tokens. Agent's decimal claim normalizes to WEI assuming 18 decimals; server decodes raw WEI. False divergences for USDC etc. when the agent claims `"100"` (the agent's literal) — server sees `100000000` WEI which decoded as 18 decimals is `0.0000001 * 10^8` etc. Wrong by 10^12 orders of magnitude.
- **(b) Synchronous decimals lookup via the bridged-variants table (Plan 08-04 SOT)**: `lookupBridgedVariant(record.tx.to.toLowerCase())` returns rows matching the address; rows have implicit decimals per the BRIDGED_VARIANTS data model (would need to add a `decimals: number` field to BridgedVariant interface — small Plan 08-04 retroactive extension). NOT a v1.3 change; v1.4+ scope.
- **(c) Always compare WEI-to-WEI; agent MUST pass WEI string in `claimedDecode.args.amount`**: shift the decimal-normalization responsibility to the agent (the agent already calls `prepare_token_send` with a decimal string + the server resolves WEI; for verify_tx_decode, the agent can read the WEI value from the prepare response and pass it back as a string). Simplest; documented in tool description.
