```typescript
// src/tools/get_verification_artifact.ts (NEW Phase 9 Plan 09-03)
//
// Sparse JSON + pasteableBlock for SECOND-LLM out-of-band verification.
// Mirror of get_tx_verification.ts handle-lookup-and-emit scaffolding,
// simplified to a single block payload (no PREPARE RECEIPT / LEDGER /
// AGENT TASK / 4byte / VERIFY BEFORE SIGNING re-emission — those live in
// get_tx_verification's user-facing ritual surface).
//
// Audience: a FRESH chat session with no shared context — second LLM is the
// decoder. The MCP provides the bytes; the second LLM tells the user what
// they mean. Defense against fully-coordinated agent compromise.
//
// NOT REGISTERED in this plan — Plan 09-05 owns the register-all.ts import
// line (carved to avoid Plan 09-03 ∥ 09-05 rebase conflict per PATTERNS.md § 3).

import { isDemoMode } from "../config/env.js";
import { lookup } from "../signing/handle-store.js";
import { PASTEABLE_BLOCK_TEMPLATE } from "../signing/blocks.js";
import { makeStructuredError } from "../signing/error-codes.js";
import { registerTool } from "./index.js";

const DESCRIPTION = [
  "Emit a sparse JSON + paste-able block for SECOND-LLM out-of-band verification of a prepared transaction.",
  "Use AFTER preview_send and BEFORE send_transaction when the user wants an independent decode from a second LLM (catches coordinated-agent compromise where this agent's narrative cannot be trusted).",
  "Returns `{ to, valueWei, data, chainId, payloadFingerprint, presignHash, selector }` as structuredContent + a `pasteableBlock` text payload bounded by explicit copy markers. Instruct the user: copy everything between the markers, paste into a fresh Claude/GPT/Gemini session, compare that LLM's decode against this conversation's narrative. Disagreement → DO NOT SIGN.",
  "Do NOT use as the only verification — the LEDGER BLIND-SIGN HASH on-device match is the trust anchor; this is defense-in-depth against a fully-coordinated agent compromise.",
  "Distinct from get_tx_verification (which re-emits the FULL Phase 4 verification ritual for context-evicted agent recovery — user-facing visual surface). This tool emits ONLY the pasteableBlock + sparse JSON for second-LLM consumption.",
  "Distinct from verify_tx_decode (which is an INLINE server-side cross-check — fast field-by-field diff). This tool is an OUT-OF-BAND ritual — user reads second LLM's verdict and decides.",
  "15-min TTL from the original prepare. Past TTL → HANDLE_EXPIRED; re-run prepare.",
].join(" ");

const INPUT_SCHEMA = {
  type: "object" as const,
  properties: {
    handle: { type: "string" },
  },
  required: ["handle"],
  additionalProperties: false,
};

registerTool("get_verification_artifact", DESCRIPTION, INPUT_SCHEMA, async (args) => {
  // 1. Demo-mode refusal FIRST — mirror of get_tx_verification.ts:76-88
  if (isDemoMode()) {
    return {
      isError: true,
      content: [{ type: "text" as const, text: "error: get_verification_artifact refuses in demo mode" }],
      structuredContent: makeStructuredError(
        "DEMO_MODE_REFUSED",
        "get_verification_artifact refuses in demo mode — second-LLM ritual is for real signing flows only",
      ),
    };
  }

  // 2. Handle lookup — mirror of get_tx_verification.ts:91-102
  const handleArg = (args as { handle: string }).handle;
  const lookupResult = lookup(handleArg);
  if (lookupResult.kind !== "found") {
    return {
      isError: true,
      content: [{ type: "text" as const, text: `error: ${lookupResult.message}` }],
      structuredContent: makeStructuredError(lookupResult.errorCode, lookupResult.message),
    };
  }
  const { record } = lookupResult;
  const pinned = record.pinned;

  // 3. Compose sparse JSON + pasteableBlock
  const presignHash = pinned?.presignHash ?? "(not yet previewed)";
  const block = PASTEABLE_BLOCK_TEMPLATE
    .replace("{TO}", record.tx.to)
    .replace("{VALUE_WEI}", record.tx.valueWei.toString())
    .replace("{DATA}", record.tx.data)
    .replace("{CHAIN_ID}", String(record.tx.chainId))
    .replace("{PAYLOAD_FINGERPRINT}", record.payloadFingerprint)
    .replace("{PRESIGN_HASH}", presignHash);

  return {
    content: [{ type: "text" as const, text: block }],
    structuredContent: {
      to: record.tx.to,
      valueWei: record.tx.valueWei.toString(),
      data: record.tx.data,
      chainId: record.tx.chainId,
      payloadFingerprint: record.payloadFingerprint,
      presignHash: pinned?.presignHash ?? null,  // null on prepared; populated on previewed/sent/cancelled
      selector: record.tx.data === "0x" ? null : record.tx.data.slice(0, 10),
    },
  };
});
```
