## test/sha256.sh shape (LOCKED)

```bash
#!/usr/bin/env bash
# vaultpilot-preflight-skill smoke test.
# Asserts the README's documented EXPECTED_SKILL_SHA256 matches the actual SKILL.md content.
set -euo pipefail

cd "$(dirname "$0")/.."

if [ ! -f SKILL.md ]; then
  echo "FAIL: SKILL.md not found at repo root"
  exit 1
fi

if [ ! -f README.md ]; then
  echo "FAIL: README.md not found at repo root"
  exit 1
fi

COMPUTED=$(sha256sum SKILL.md | awk '{print $1}')
DOCUMENTED=$(grep -oE 'Expected SHA-256: `[a-f0-9]+`' README.md | head -1 | sed -E 's/.*`([a-f0-9]+)`.*/\1/')

if [ -z "$DOCUMENTED" ]; then
  echo "FAIL: README.md does not contain 'Expected SHA-256: \`<hex>\`'"
  exit 1
fi

if [ "$COMPUTED" != "$DOCUMENTED" ]; then
  echo "FAIL: SKILL.md SHA-256 mismatch"
  echo "  Computed:   $COMPUTED"
  echo "  Documented: $DOCUMENTED"
  exit 1
fi

echo "OK: SKILL.md SHA-256 matches README.md ($COMPUTED)"
```
