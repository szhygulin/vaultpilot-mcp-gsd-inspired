From src/signing/payload-fingerprint-solana.ts (the fingerprint sibling — SAME hash function, DIFFERENT domain tag):
```typescript
import { concat, keccak256, toBytes } from "viem";
import type { Hex } from "viem";
export const FINGERPRINT_DOMAIN_TAG_SOLANA = "VaultPilot-soltx-v1:"; // 20 UTF-8 bytes
export function computeSolanaPayloadFingerprint(input: { messageBytes: Uint8Array }): Hex {
  const tag = toBytes(FINGERPRINT_DOMAIN_TAG_SOLANA);
  return keccak256(concat([tag, input.messageBytes]));
}
export const _solanaFingerprint = { computeSolanaPayloadFingerprint };
```
