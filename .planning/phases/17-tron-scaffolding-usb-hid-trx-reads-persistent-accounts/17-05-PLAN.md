     if (persona !== null && (!persona || typeof persona.tronAddress !== "string")) {
       throw new TypeError("setActiveTronPersona: persona.tronAddress must be a string");
     }
     activeTronPersona = persona;
   }

   export function setActiveTronPersonaBySlug(slug: string): TronPersona {
     const found = findTronPersona(slug as TronPersonaSlug);
     if (!found) throw new Error(`Unknown TRON persona slug: ${slug}`);

       slug: found.slug,
       tronAddress: found.tronAddress,
       description: found.description,
     };
     return activeTronPersona;
   }
