2. **`src/tokens/solana-top-50.ts` (new)** — typed loader:
   ```
   import raw from "./solana-top-50.json" with { type: "json" };
   import { PublicKey } from "@solana/web3.js";
   
   export interface SolanaTokenRegistryEntry {
     mint: string;
     symbol: string;
     decimals: number;
     displayName: string;
   }
   
   // DOA validation at module load — invalid base58 throws here, not at runtime.
   const validated: SolanaTokenRegistryEntry[] = raw.map((e) => {
     new PublicKey(e.mint);  // throws on malformed base58 → DOA at import time
     return e;
   });
   
   const byMint = new Map(validated.map((e) => [e.mint, e]));
   
   export function findByMint(mint: string): SolanaTokenRegistryEntry | undefined {
     return byMint.get(mint);
   }
   
   export function listSolanaTokens(): readonly SolanaTokenRegistryEntry[] {
     return validated;
   }
   ```
